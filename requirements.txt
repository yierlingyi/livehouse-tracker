# ---- Web 框架与 ASGI 服务器 ----
fastapi                   # backend/*.py / middleware 直接使用（APIRouter/FastAPI/Depends）
uvicorn[standard]         # 生产 ASGI 服务器（standard 含 uvloop/httptools 等 Linux 加速）

# ---- 数据库驱动 ----
asyncpg                   # backend/main.py 连接池 + 各 API service 查询

# ---- 文件上传（UploadFile/File 依赖）----
python-multipart          # backend/api/upload.py

# ---- 缓存 / 限流（可选，仅在配置 REDIS_URL 时使用）----
redis                     # backend/services/cache.py、rate_limiter.py 惰性导入

# ---- 业务时区（zoneinfo 数据，缺失时回落 UTC；最小化 Linux 镜像建议安装）----
tzdata                    # backend/api/full.py: business_today() 使用 zoneinfo.ZoneInfo
